과제 1 · [난이도 중] 전위·후위 ++ / --와 += 오버로딩 — Time 클래스

5주차의 Time 클래스에 증가·감소 연산자를 추가합니다. ++와 --는 1초씩 바꾸고, 자정을 넘으면 순환합니다(23:59:59 → 00:00:00, 00:00:00 → 23:59:59). +=는 지정한 초만큼 시각을 늘립니다.

조건: 멤버는 unsigned int hour, minute, second · 생성자 explicit Time( int = 0, int = 0, int = 0 )은 범위를 벗어나면 invalid_argument · 전위 Time &operator++(), 후위 Time operator++( int ), 전위 Time &operator--(), 후위 Time operator--( int ) · Time &operator+=( unsigned int ) — 연쇄할 수 있어야 한다 · 1초 증가·감소 처리는 private 유틸리티 함수 tick(), untick()에 두고 연산자 함수들은 이 함수를 호출만 한다(교재 helpIncrement와 같은 구조) · 후위 연산자는 증가 전 상태를 temp에 저장해 값으로 반환 · 출력은 friend operator<<로 HH:MM:SS 형식 · main에서 시·분·초와 더할 초를 입력받아 처리하고, 숫자가 아닌 입력과 음수 초를 검사할 것

t1++ = 23:59:59

t2 = 00:00:00
--t2 = 23:59:59
t2-- = 23:59:59
t2 = 23:59:58

t3 += 95 -> 11:00:05
(t3 += 30) += 30 -> 11:01:05

// 과제 1 · 전위·후위 ++ / -- 와 += 오버로딩 — Time 클래스 (교재 그림 9.24 Time + 10.7 기반)
#include <iostream>
#include <iomanip>
#include <stdexcept>
using namespace std;

class Time
{
   friend ostream &operator<<( ostream &, const Time & );
public:
   explicit Time( int = 0, int = 0, int = 0 );
   Time &setTime( int, int, int );

   Time &operator++();          // 전위 ++ : 1초 증가
   Time operator++( int );      // 후위 ++
   Time &operator--();          // 전위 -- : 1초 감소
   Time operator--( int );      // 후위 --
   Time &operator+=( unsigned int ); // 지정한 초만큼 증가
private:
   unsigned int hour = 0;   // 0 - 23
   unsigned int minute = 0; // 0 - 59
   unsigned int second = 0; // 0 - 59
   void tick();     // 1초 증가 (23:59:59 → 00:00:00)
   void untick();   // 1초 감소 (00:00:00 → 23:59:59)
};

Time::Time( int h, int m, int s )
{
   setTime( h, m, s );
}

Time &Time::setTime( int h, int m, int s )
{
   if ( h < 0 || h > 23 )
      throw invalid_argument( "hour must be 0-23" );
   if ( m < 0 || m > 59 )
      throw invalid_argument( "minute must be 0-59" );
   if ( s < 0 || s > 59 )
      throw invalid_argument( "second must be 0-59" );
   hour = h;
   minute = m;
   second = s;
   return *this;
}

void Time::tick()
{
   if ( second < 59 )
      ++second;
   else
   {
      second = 0;
      if ( minute < 59 )
         ++minute;
      else
      {
         minute = 0;
         hour = ( hour < 23 ) ? hour + 1 : 0;
      }
   }
}

void Time::untick()
{
   if ( second > 0 )
      --second;
   else
   {
      second = 59;
      if ( minute > 0 )
         --minute;
      else
      {
         minute = 59;
         hour = ( hour > 0 ) ? hour - 1 : 23;
      }
   }
}

Time &Time::operator++()
{
   tick();
   return *this;      // 증가된 자기 자신을 참조로 반환
}

Time Time::operator++( int )
{
   Time temp = *this; // 증가 전 상태 저장
   tick();
   return temp;       // 증가 전 값을 값으로 반환
}

Time &Time::operator--()
{
   untick();
   return *this;
}

Time Time::operator--( int )
{
   Time temp = *this;
   untick();
   return temp;
}

Time &Time::operator+=( unsigned int seconds )
{
   for ( unsigned int i = 0; i < seconds; ++i )
      tick();
   return *this;      // 연쇄를 위해 참조 반환
}

ostream &operator<<( ostream &output, const Time &t )
{
   output << setfill( '0' ) << setw( 2 ) << t.hour << ":"
      << setw( 2 ) << t.minute << ":" << setw( 2 ) << t.second << setfill( ' ' );
   return output;
}

int main()
{
   Time t1( 23, 59, 58 );
   cout << "t1 = " << t1 << endl;
   cout << "++t1 = " << ++t1 << endl;
   cout << "t1++ = " << t1++ << endl;
   cout << "t1 = " << t1 << "\n\n";

   Time t2;
   cout << "t2 = " << t2 << endl;
   cout << "--t2 = " << --t2 << endl;
   cout << "t2-- = " << t2-- << endl;
   cout << "t2 = " << t2 << "\n\n";

   Time t3( 10, 58, 30 );
   cout << "t3 = " << t3 << endl;
   cout << "t3 += 95 -> " << ( t3 += 95 ) << endl;
   cout << "(t3 += 30) += 30 -> " << ( ( t3 += 30 ) += 30 ) << "\n\n";

   int h, m, s, add;
   cout << "시, 분, 초와 더할 초를 입력하세요: ";
   if ( !( cin >> h >> m >> s >> add ) )
   {
      cout << "숫자를 입력해야 합니다." << endl;
      return 1;
   }
   if ( add < 0 )
   {
      cout << "더할 초는 0 이상이어야 합니다." << endl;
      return 1;
   }
   try
   {
      Time t4( h, m, s );
      cout << "입력 시각: " << t4 << endl;
      cout << add << "초 후: " << ( t4 += static_cast< unsigned int >( add ) ) << endl;
   }
   catch ( invalid_argument &e )
   {
      cout << "예외 발생: " << e.what() << endl;
   }
}

실행 예시 ①의 ++t1과 t1++은 둘 다 23:59:59를 출력하지만 의미가 다릅니다. ++t1은 23:59:58을 증가시킨 뒤의 값이고, t1++은 23:59:59를 증가시키기 전의 값입니다. 그 결과 다음 줄에서 t1이 00:00:00이 되었습니다. 증가·감소 규칙을 tick()과 untick() 한 곳에만 두었기 때문에 ++, ++(int), += 세 연산자가 같은 규칙을 공유합니다. 더할 초를 int로 받아 음수를 먼저 거르는 이유는, unsigned int로 바로 받으면 -5가 매우 큰 양수로 바뀌어 검사를 통과해 버리기 때문입니다.

과제 2 · [난이도 중] Date 클래스 확장 — 전위·후위 --와 관계 연산자 6개

교재 그림 10.6~10.8의 Date 클래스를 그대로 가져와서 감소 연산자와 비교 연산자를 추가합니다. 날짜를 하루 줄이는 것은 늘리는 것보다 까다롭습니다. 이전 달의 마지막 날이 며칠인지 알아야 하기 때문입니다.

조건: 교재의 setDate, operator++(전위·후위), operator+=, leapYear, endOfMonth, helpIncrement, operator<<는 그대로 둔다 · 추가할 것: 전위 Date &operator--(), 후위 Date operator--( int ), private 유틸리티 함수 helpDecrement() · 해당 월의 마지막 날을 윤년까지 반영해 돌려주는 private 함수 lastDay( unsigned int month ) const를 만들어 helpDecrement에서 사용 · 1900년 1월 1일에서 감소시키면 out_of_range 예외를 던지고 객체는 바꾸지 않는다 · 관계 연산자 ==, !=, <, >, <=, >=를 모두 const 멤버 함수로 만든다 — 단 직접 비교하는 것은 <와 == 둘뿐이고, 나머지 넷은 이 둘을 호출해 작성한다 · 비교는 연 → 월 → 일 순서

d3 = December 31, 2010

// 과제 2 · 전위·후위 -- 와 관계 연산자 6개 오버로딩 — Date 클래스 확장 (교재 그림 10.6~10.8 기반)
#include <array>
#include <iostream>
#include <string>
#include <stdexcept>
using namespace std;

class Date
{
   friend ostream &operator<<( ostream &, const Date & );
public:
   Date( int m = 1, int d = 1, int y = 1900 );
   void setDate( int, int, int );
   Date &operator++();
   Date operator++( int );
   Date &operator+=( unsigned int );

   // ---- 이번 과제에서 추가 ----
   Date &operator--();          // 전위 --
   Date operator--( int );      // 후위 --
   bool operator==( const Date & ) const;
   bool operator!=( const Date & ) const;
   bool operator<( const Date & ) const;
   bool operator>( const Date & ) const;
   bool operator<=( const Date & ) const;
   bool operator>=( const Date & ) const;

   static bool leapYear( int );
   bool endOfMonth( int ) const;
private:
   unsigned int month;
   unsigned int day;
   unsigned int year;

   static const array< unsigned int, 13 > days;
   void helpIncrement();
   void helpDecrement();                    // 추가
   unsigned int lastDay( unsigned int ) const; // 추가: 해당 월의 마지막 날
};

const array< unsigned int, 13 > Date::days =
   { 0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31 };

Date::Date( int month, int day, int year )
{
   setDate( month, day, year );
}

void Date::setDate( int mm, int dd, int yy )
{
   if ( mm >= 1 && mm <= 12 )
      month = mm;
   else
      throw invalid_argument( "Month must be 1-12" );

   if ( yy >= 1900 && yy <= 2100 )
      year = yy;
   else
      throw invalid_argument( "Year must be >= 1900 and <= 2100" );

   if ( ( month == 2 && leapYear( year ) && dd >= 1 && dd <= 29 ) ||
        ( dd >= 1 && dd <= static_cast< int >( days[ month ] ) ) )
      day = dd;
   else
      throw invalid_argument( "Day is out of range for current month and year" );
}

Date &Date::operator++()
{
   helpIncrement();
   return *this;
}

Date Date::operator++( int )
{
   Date temp = *this;
   helpIncrement();
   return temp;
}

Date &Date::operator+=( unsigned int additionalDays )
{
   for ( unsigned int i = 0; i < additionalDays; ++i )
      helpIncrement();
   return *this;
}

Date &Date::operator--()
{
   helpDecrement();
   return *this;
}

Date Date::operator--( int )
{
   Date temp = *this;
   helpDecrement();
   return temp;
}

// 연 → 월 → 일 순서로 비교
bool Date::operator<( const Date &right ) const
{
   if ( year != right.year )
      return year < right.year;
   if ( month != right.month )
      return month < right.month;
   return day < right.day;
}

bool Date::operator==( const Date &right ) const
{
   return year == right.year && month == right.month && day == right.day;
}

// 나머지 넷은 < 와 == 를 이용해 작성 — 규칙이 한 곳에만 있도록
bool Date::operator!=( const Date &right ) const { return !( *this == right ); }
bool Date::operator>( const Date &right ) const { return right < *this; }
bool Date::operator<=( const Date &right ) const { return !( right < *this ); }
bool Date::operator>=( const Date &right ) const { return !( *this < right ); }

bool Date::leapYear( int testYear )
{
   return testYear % 400 == 0 || ( testYear % 100 != 0 && testYear % 4 == 0 );
}

unsigned int Date::lastDay( unsigned int m ) const
{
   return ( m == 2 && leapYear( year ) ) ? 29 : days[ m ];
}

bool Date::endOfMonth( int testDay ) const
{
   return testDay == static_cast< int >( lastDay( month ) );
}

void Date::helpIncrement()
{
   if ( !endOfMonth( day ) )
      ++day;
   else if ( month < 12 )
   {
      ++month;
      day = 1;
   }
   else
   {
      ++year;
      month = 1;
      day = 1;
   }
}

void Date::helpDecrement()
{
   if ( day > 1 )
      --day;
   else if ( month > 1 )
   {
      --month;
      day = lastDay( month ); // 이전 달의 마지막 날 (윤년 반영)
   }
   else
   {
      if ( year == 1900 )
         throw out_of_range( "Date cannot go before January 1, 1900" );
      --year;
      month = 12;
      day = 31;
   }
}

ostream &operator<<( ostream &output, const Date &d )
{
   static string monthName[ 13 ] = { "", "January", "February", "March", "April",
      "May", "June", "July", "August", "September", "October", "November", "December" };
   output << monthName[ d.month ] << ' ' << d.day << ", " << d.year;
   return output;
}

int main()
{
   Date d1( 3, 1, 2024 );
   cout << "d1 = " << d1 << endl;
   cout << "--d1 = " << --d1 << "  (2024년은 윤년)" << endl;
   Date d2( 3, 1, 2100 );
   cout << "--d2 = " << --d2 << "  (2100년은 윤년 아님)" << endl;
   Date d3( 1, 1, 2011 );
   cout << "d3-- = " << d3-- << endl;
   cout << "d3 = " << d3 << "\n\n";

   Date a( 7, 14, 2010 ), b( 7, 15, 2010 ), c( 7, 14, 2010 );
   cout << boolalpha;
   cout << "a = " << a << ", b = " << b << ", c = " << c << endl;
   cout << "a == c : " << ( a == c ) << "   a != b : " << ( a != b ) << endl;
   cout << "a <  b : " << ( a < b ) << "   a >  b : " << ( a > b ) << endl;
   cout << "a <= c : " << ( a <= c ) << "   b >= a : " << ( b >= a ) << endl;
   cout << "++a == b : " << ( ++a == b ) << "\n\n";

   Date first;
   cout << "first = " << first << endl;
   try
   {
      --first;
   }
   catch ( out_of_range &e )
   {
      cout << "예외 발생: " << e.what() << endl;
   }
   cout << "first = " << first << endl;
}

3월 1일에서 하루를 빼는 경우가 핵심입니다. 2024년은 윤년이라 2월 29일, 2100년은 평년이라 2월 28일이 되며, 이것을 lastDay() 한 곳에서 처리합니다. 관계 연산자를 <와 ==만 직접 구현한 이유는 비교 규칙을 한 곳에 두기 위해서입니다. 예를 들어 a > b는 b < a이고 a <= b는 !( b < a )입니다. 여섯 개를 따로 구현하면 그중 하나만 실수해도 서로 결과가 어긋날 수 있습니다(10.3 '짝을 맞춘다' 원칙). 마지막 줄에서는 예외가 났는데도 first가 그대로입니다. helpDecrement가 날짜를 바꾸기 전에 검사하고 예외를 던졌기 때문입니다.

과제 3 · [난이도 상] 날짜 계산기 — 이항 + / -와 >> 오버로딩

교재 Date 클래스로 날짜 계산기를 만듭니다. 'n일 뒤의 날짜'와 '두 날짜 사이의 일수'를 연산자로 계산하고, 날짜를 MM/DD/YYYY 형식으로 입력받습니다. 지금까지 배운 연산자 오버로딩을 모두 사용하는 종합 과제입니다.

조건: 교재 Date의 멤버 함수와 friend operator<<는 그대로 둔다 · Date operator+( unsigned int ) const — +=에 위임해서 작성하고, 원래 객체는 바꾸지 않고 새 Date를 값으로 반환 · Date operator+( unsigned int, const Date & ) — n + 날짜 형태를 위한 비멤버 함수(private에 접근할 필요가 없으면 friend로 만들지 않는다) · int operator-( const Date & ) const — 두 날짜 사이의 일수, 왼쪽 날짜가 더 이르면 음수 · 이를 위해 ==와 <도 정의 · friend operator>> — MM/DD/YYYY 형식으로 읽는다. 구분자가 /가 아니거나 숫자가 아니면 failbit를 세우고(형식 오류), 범위 검사는 setDate에 맡겨 invalid_argument가 그대로 전파되게 한다(값 오류) · main에서 날짜와 더할 일 수를 입력받아 start + n, n + start, 두 방향의 뺄셈을 출력하고, 계산 뒤에도 start가 바뀌지 않았는지 확인 · 더할 일 수가 음수이면 거부

실행 예시 ① — 교재 그림 10.8과 같은 계산

날짜(MM/DD/YYYY)와 더할 일 수를 입력하세요: 12/27/2010 7
December 27, 2010 + 7 = January 3, 2011
7 + December 27, 2010 = January 3, 2011
start는 그대로: December 27, 2010
later - start = 7일
start - later = -7일

366 + January 1, 2024 = January 1, 2025
start는 그대로: January 1, 2024

// 과제 3 · 이항 산술 연산자와 >> 오버로딩 — 날짜 계산기 (교재 그림 10.6~10.8 + 10.5 기반)
#include <array>
#include <iostream>
#include <string>
#include <stdexcept>
using namespace std;

class Date
{
   friend ostream &operator<<( ostream &, const Date & );
   friend istream &operator>>( istream &, Date & );
public:
   Date( int m = 1, int d = 1, int y = 1900 );
   void setDate( int, int, int );
   Date &operator++();
   Date operator++( int );
   Date &operator+=( unsigned int );

   // ---- 이번 과제에서 추가 ----
   Date operator+( unsigned int ) const;   // Date + n  (값 반환)
   int operator-( const Date & ) const;    // Date - Date → 날짜 차이(일)
   bool operator==( const Date & ) const;
   bool operator<( const Date & ) const;

   static bool leapYear( int );
   bool endOfMonth( int ) const;
private:
   unsigned int month;
   unsigned int day;
   unsigned int year;
   static const array< unsigned int, 13 > days;
   void helpIncrement();
};

Date operator+( unsigned int, const Date & ); // n + Date  (비멤버)

const array< unsigned int, 13 > Date::days =
   { 0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31 };

Date::Date( int month, int day, int year )
{
   setDate( month, day, year );
}

void Date::setDate( int mm, int dd, int yy )
{
   if ( mm >= 1 && mm <= 12 )
      month = mm;
   else
      throw invalid_argument( "Month must be 1-12" );

   if ( yy >= 1900 && yy <= 2100 )
      year = yy;
   else
      throw invalid_argument( "Year must be >= 1900 and <= 2100" );

   if ( ( month == 2 && leapYear( year ) && dd >= 1 && dd <= 29 ) ||
        ( dd >= 1 && dd <= static_cast< int >( days[ month ] ) ) )
      day = dd;
   else
      throw invalid_argument( "Day is out of range for current month and year" );
}

Date &Date::operator++()
{
   helpIncrement();
   return *this;
}

Date Date::operator++( int )
{
   Date temp = *this;
   helpIncrement();
   return temp;
}

Date &Date::operator+=( unsigned int additionalDays )
{
   for ( unsigned int i = 0; i < additionalDays; ++i )
      helpIncrement();
   return *this;
}

// + 는 += 에 위임한다 — 복사본을 만들어 늘린 뒤 값으로 반환
Date Date::operator+( unsigned int n ) const
{
   Date result = *this;
   result += n;
   return result;
}

// 왼쪽 피연산자가 Date가 아니므로 비멤버. private 접근이 필요 없어 friend도 아니다
Date operator+( unsigned int n, const Date &d )
{
   return d + n;
}

bool Date::operator==( const Date &right ) const
{
   return year == right.year && month == right.month && day == right.day;
}

bool Date::operator<( const Date &right ) const
{
   if ( year != right.year )
      return year < right.year;
   if ( month != right.month )
      return month < right.month;
   return day < right.day;
}

// 이른 날짜에서 늦은 날짜까지 하루씩 세어 차이를 구한다 (this가 이르면 음수)
int Date::operator-( const Date &right ) const
{
   bool negative = *this < right;
   Date from = negative ? *this : right;
   const Date &to = negative ? right : *this;

   int count = 0;
   while ( !( from == to ) )
   {
      ++from;
      ++count;
   }
   return negative ? -count : count;
}

bool Date::leapYear( int testYear )
{
   return testYear % 400 == 0 || ( testYear % 100 != 0 && testYear % 4 == 0 );
}

bool Date::endOfMonth( int testDay ) const
{
   if ( month == 2 && leapYear( year ) )
      return testDay == 29;
   else
      return testDay == static_cast< int >( days[ month ] );
}

void Date::helpIncrement()
{
   if ( !endOfMonth( day ) )
      ++day;
   else if ( month < 12 )
   {
      ++month;
      day = 1;
   }
   else
   {
      ++year;
      month = 1;
      day = 1;
   }
}

ostream &operator<<( ostream &output, const Date &d )
{
   static string monthName[ 13 ] = { "", "January", "February", "March", "April",
      "May", "June", "July", "August", "September", "October", "November", "December" };
   output << monthName[ d.month ] << ' ' << d.day << ", " << d.year;
   return output;
}

// MM/DD/YYYY 형식으로 입력
//  - 구분자가 '/'가 아니거나 숫자가 아니면 failbit (형식 오류)
//  - 범위 밖이면 setDate가 던지는 invalid_argument가 전파 (값 오류)
istream &operator>>( istream &input, Date &d )
{
   int m, dd, y;
   char s1, s2;
   if ( input >> m >> s1 >> dd >> s2 >> y && s1 == '/' && s2 == '/' )
      d.setDate( m, dd, y );
   else
      input.setstate( ios::failbit );
   return input;
}

int main()
{
   Date start;
   int n;

   cout << "날짜(MM/DD/YYYY)와 더할 일 수를 입력하세요: ";
   try
   {
      if ( !( cin >> start >> n ) )
      {
         cout << "형식이 올바르지 않습니다." << endl;
         return 1;
      }
      if ( n < 0 )
      {
         cout << "더할 일 수는 0 이상이어야 합니다." << endl;
         return 1;
      }
      unsigned int addDays = static_cast< unsigned int >( n );

      Date later = start + addDays;
      Date later2 = addDays + start;   // 비멤버 operator+
      cout << start << " + " << n << " = " << later << endl;
      cout << n << " + " << start << " = " << later2 << endl;
      cout << "start는 그대로: " << start << endl;
      cout << "later - start = " << ( later - start ) << "일" << endl;
      cout << "start - later = " << ( start - later ) << "일" << endl;
   }
   catch ( invalid_argument &e )
   {
      cout << "예외 발생: " << e.what() << endl;
   }
}

세 가지를 확인해 보세요. 첫째, +는 +=에 위임했습니다. 복사본을 만들어 +=를 적용한 뒤 그 복사본을 값으로 반환하므로 원래 객체는 바뀌지 않습니다. 실행 예시마다 'start는 그대로'가 이것을 보여 줍니다(10.3 '+는 +=에 위임한다' 원칙). 둘째, n + start는 왼쪽 피연산자가 unsigned int라서 멤버 함수가 될 수 없습니다. 그래서 비멤버로 만들었고, 내부에서 public 멤버인 d + n을 호출하므로 friend일 필요도 없습니다. 셋째, 뺄셈은 같은 Date끼리 연산하지만 결과는 Date가 아니라 일수(int)입니다. 연산자의 반환형은 피연산자와 같아야 하는 것이 아니라, 그 연산의 의미에 맞게 정합니다.
