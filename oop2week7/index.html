과제 1 · [난이도 중] 동적 메모리 할당 기초 — new / new[] / 값 초기화 / delete[]

실행 시간에 크기가 정해지는 배열을 동적 할당해 사용하고, 다 쓴 뒤 반드시 해제합니다. new int 와 new int(), new int[10] 와 new int[10]() 의 차이를 직접 확인합니다.

조건: new int 와 new int() 를 각각 만들어 값 초기화 차이를 확인한다 — 초기화하지 않은 new int 의 값은 읽지 않는다(쓰레기값) · 배열의 크기를 입력받아 new int[ n ]() 로 할당하고, 할당 직후 모든 원소가 0인지 출력해 확인 · C++11 목록 초기화자 new int[ 5 ]{ 1, 2, 3 } 로 일부만 초기화하고 나머지가 0인지 확인 · 입력받은 정수들의 합계·평균·최댓값을 구한다(평균은 실수, 소수점 둘째 자리) · 크기가 1 이상이 아니거나 숫자가 아닌 입력은 거부 · new[] 로 할당한 배열은 반드시 delete[] 로 해제한다 — 중간에 return 하는 경우에도 해제할 것

// 과제 1 · 동적 메모리 할당 기초 — new / new[] / 값 초기화 / delete[]
#include <iostream>
#include <iomanip>
using namespace std;

int main()
{
   // 1) new int  와  new int()  의 차이
   int *noInit = new int;      // 값 초기화하지 않음 → 쓰레기값 (읽지 않는다)
   int *zeroed = new int();    // 값 초기화 → 0
   cout << "new int()  : " << *zeroed << "   (값 초기화)\n";
   cout << "new int    : 값 초기화하지 않으므로 읽지 않음 (쓰레기값)\n\n";
   delete noInit;
   delete zeroed;

   // 2) 배열 동적 할당 — 원소를 0으로 값 초기화
   int n;
   cout << "배열의 크기를 입력하세요: ";
   if ( !( cin >> n ) || n <= 0 )
   {
      cout << "1 이상의 정수를 입력해야 합니다." << endl;
      return 1;
   }

   int *grades = new int[ n ]();   // () → 모든 원소를 0으로 초기화
   cout << "할당 직후 (모두 0):";
   for ( int i = 0; i < n; ++i )
      cout << setw( 5 ) << grades[ i ];
   cout << "\n\n";

   // 3) 목록 초기화자로 일부만 지정 (C++11)
   int *fixedArr = new int[ 5 ]{ 1, 2, 3 };  // 나머지 두 원소는 0
   cout << "new int[5]{ 1, 2, 3 } :";
   for ( int i = 0; i < 5; ++i )
      cout << setw( 5 ) << fixedArr[ i ];
   cout << "\n\n";
   delete[] fixedArr;   // new[] 로 할당했으므로 delete[]

   // 4) 값을 입력받아 합·평균·최댓값 계산
   cout << n << "개의 정수를 입력하세요: ";
   for ( int i = 0; i < n; ++i )
   {
      if ( !( cin >> grades[ i ] ) )
      {
         cout << "숫자를 입력해야 합니다." << endl;
         delete[] grades;
         return 1;
      }
   }

   long long sum = 0;
   int maxValue = grades[ 0 ];
   for ( int i = 0; i < n; ++i )
   {
      sum += grades[ i ];
      if ( grades[ i ] > maxValue )
         maxValue = grades[ i ];
   }

   cout << "합계   : " << sum << endl;
   cout << "평균   : " << fixed << setprecision( 2 )
        << static_cast< double >( sum ) / n << endl;
   cout << "최댓값 : " << maxValue << "\n\n";

   delete[] grades;   // 다 쓴 뒤 반드시 해제 — new[] 와 짝
   cout << "메모리를 해제했습니다." << endl;
}

new int[ n ]() 의 괄호가 핵심입니다. 괄호가 없으면 원소가 초기화되지 않아 '할당 직후 (모두 0)' 부분에 쓰레기값이 나옵니다. 괄호를 빼고 실행해 보면 차이를 바로 볼 수 있습니다. 또 new int 는 초기화하지 않으므로 그 값을 출력하면 안 됩니다(미정의 동작). 그래서 프로그램에서도 값을 읽지 않고 설명만 출력했습니다. 해제는 new[] 로 했으므로 delete[] 입니다. 입력이 잘못돼 중간에 return 할 때도 delete[] 로 먼저 해제한 뒤 돌아가도록 했습니다 — 이렇게 하지 않으면 메모리 누수가 됩니다.

과제 2 · [난이도 중] Array 클래스 — 동적 할당과 깊은 복사

교재 그림 10.10~10.11의 Array 클래스 중 복사 생성자·소멸자·operator[]·operator<< 를 구현합니다. 포인터 멤버를 가진 클래스에서 깊은 복사가 왜 필요한지 직접 확인합니다.

조건: private 멤버 std::size_t size 와 int *ptr · 기본 생성자 explicit Array( int = 10 ) — new int[ size ]() 로 원소를 0으로 초기화 · 복사 생성자 Array( const Array & ) — 새 메모리를 할당하고 원소를 하나씩 복사하는 깊은 복사. 매개변수는 반드시 const 참조 · 소멸자에서 delete[] ptr · size_t getSize() const · operator[] 를 두 벌 만든다 — 비const는 int & 반환(수정 가능), const는 int 반환. 둘 다 범위를 벗어나면 out_of_range 예외 · friend operator<< 로 배열 전체를 출력 · 생성자·복사 생성자·소멸자에서 호출 메시지를 출력해 언제 호출되는지 관찰 · main에서 ① 기본 생성 ② a[i] = 값 으로 채우기 ③ 복사 생성(Array b = a;) 후 b만 수정해 a가 안 바뀌는지 확인 ④ 함수에 값으로 전달 ⑤ 함수에서 값으로 반환 ⑥ 범위를 벗어난 첨자

// 과제 2 · Array 클래스 — 동적 할당과 깊은 복사
#include <iostream>
#include <iomanip>
#include <stdexcept>
#include "task2_array_deep_copy.h"
using namespace std;

const size_t Array::sizeDefault;

// 기본 생성자: size개의 원소를 0으로 초기화해 동적 할당
Array::Array( int arraySize )
   : size( arraySize < 0 ? 0 : arraySize ),
     ptr( size > 0 ? new int[ size ]() : nullptr )  // () 로 값 초기화 → 0
{
   // new int[ size ]() 가 모든 원소를 0으로 초기화하므로 따로 반복할 필요가 없다
   cout << "  [Array 생성자] size=" << size << endl;
}

// 복사 생성자: 새 메모리를 할당하고 원소를 하나씩 복사한다 (깊은 복사)
//   반드시 참조로 받아야 한다. 값으로 받으면 복사본을 만들기 위해 복사 생성자를
//   다시 호출하게 되어 무한 재귀가 된다.
Array::Array( const Array &arrayToCopy )
   : size( arrayToCopy.size ),
     ptr( size > 0 ? new int[ size ] : nullptr )
{
   for ( size_t i = 0; i < size; ++i )
      ptr[ i ] = arrayToCopy.ptr[ i ];
   cout << "  [Array 복사 생성자] size=" << size << " (깊은 복사)" << endl;
}

// 소멸자: new[] 로 할당했으므로 delete[] 로 해제
Array::~Array()
{
   cout << "  [Array 소멸자] size=" << size << " 메모리 해제" << endl;
   delete[] ptr;
}

size_t Array::getSize() const { return size; }

// 비const 객체용: 참조를 반환하므로 array[i] = 값; 이 가능
int &Array::operator[]( int subscript )
{
   if ( subscript < 0 || subscript >= static_cast< int >( size ) )
      throw out_of_range( "Subscript out of range" );
   return ptr[ subscript ];
}

// const 객체용: 값을 반환하므로 읽기만 가능
int Array::operator[]( int subscript ) const
{
   if ( subscript < 0 || subscript >= static_cast< int >( size ) )
      throw out_of_range( "Subscript out of range" );
   return ptr[ subscript ];
}

ostream &operator<<( ostream &output, const Array &a )
{
   for ( size_t i = 0; i < a.size; ++i )
   {
      output << setw( 8 ) << a.ptr[ i ];
      if ( ( i + 1 ) % 4 == 0 )
         output << endl;
   }
   if ( a.size % 4 != 0 )
      output << endl;
   return output;
}

// 값으로 받아 복사 생성자가 호출되는지 확인하는 함수
void printByValue( Array a )
{
   cout << "printByValue 안에서 (값 전달):\n" << a;
}

// 값으로 반환해 복사 생성자가 호출되는지 확인하는 함수
Array makeFilled( size_t n, int value )
{
   Array temp( static_cast< int >( n ) );
   for ( size_t i = 0; i < n; ++i )
      temp[ static_cast< int >( i ) ] = value + static_cast< int >( i );
   return temp;
}

int main()
{
   cout << "--- 1. 기본 생성 ---\n";
   Array a( 5 );

   cout << "\n--- 2. operator[] 로 값 채우기 ---\n";
   for ( int i = 0; i < 5; ++i )
      a[ i ] = ( i + 1 ) * 10;
   cout << a;

   cout << "\n--- 3. 복사 생성 (선언과 함께 = 를 쓴 경우) ---\n";
   Array b = a;              // 대입이 아니라 복사 생성자 호출
   b[ 0 ] = 999;
   cout << "a:\n" << a;
   cout << "b (b[0]만 바꿈):\n" << b;

   cout << "\n--- 4. 함수에 값으로 전달 ---\n";
   printByValue( a );

   cout << "\n--- 5. 함수에서 값으로 반환 ---\n";
   Array c = makeFilled( 4, 100 );
   cout << "반환받은 c:\n" << c;

   cout << "\n--- 6. 범위를 벗어난 첨자 ---\n";
   try
   {
      cout << "a[ 10 ] 을 실행합니다." << endl;
      a[ 10 ] = 1;
   }
   catch ( out_of_range &e )
   {
      cout << "예외 발생: " << e.what() << endl;
   }

   cout << "\n--- 7. main 종료 (소멸 순서 확인) ---\n";
}

세 가지를 확인해 보세요. 첫째, 3번에서 Array b = a; 는 대입이 아니라 복사 생성자 호출이고, b[0]만 999로 바꿨는데 a는 그대로입니다 — 깊은 복사가 되었기 때문입니다(깊은 복사가 아니면 둘 다 999가 됩니다). 둘째, 4번처럼 함수에 값으로 전달할 때도 복사 생성자가 호출됩니다. 셋째, 5번에서는 [Array 생성자] size=4 만 보이고 복사 생성자 메시지가 없습니다. 교재는 값을 반환할 때 복사 생성자가 호출된다고 설명하지만, 요즘 컴파일러는 복사 생략(copy elision) 으로 이 복사를 없애 버립니다. 복사 생성자가 필요 없다는 뜻이 아니라, 컴파일러가 임시 객체를 곧바로 c 자리에 만들어 복사를 아예 하지 않는 것입니다(7번에서 소멸자가 한 번만 찍히는 것으로 확인됩니다).

과제 3 · [난이도 상] Array 클래스 완성 — 복사 대입, == / !=, >> 입력, explicit와 static_cast

과제 2의 Array 클래스에 복사 대입 연산자와 관계 연산자, 스트림 입력 연산자를 추가해 교재 그림 10.9의 테스트 프로그램이 동작하게 만듭니다. 마지막으로 explicit 생성자와 static_cast 를 확인합니다.

조건: 과제 2의 멤버와 함수를 그대로 두고 다음을 추가 · 복사 대입 연산자 const Array &operator=( const Array & ) — ① if ( &right != this ) 자기 대입 검사 ② 크기가 다르면 기존 메모리를 delete[] 하고 새 크기로 할당 ③ 원소를 하나씩 복사 ④ return *this; 로 연쇄 대입 가능 · operator== — 먼저 원소 개수를 비교하고, 같으면 원소를 차례로 비교 · operator!= 는 operator== 를 뒤집어 작성 · friend operator>> 로 배열 전체 입력 · explicit Array( int = 10 ) 를 유지하고, const Array & 를 받는 outputArray 함수를 만들어 outputArray( 3 ) 은 컴파일 오류가 되고 outputArray( static_cast< Array >( 3 ) ) 은 동작함을 확인(전자는 주석으로 남길 것) · main에서 크기가 다른 Array끼리 대입, 자기 대입, 연쇄 대입, == 와 !=, 깊은 복사 확인, 범위 밖 첨자까지 확인

// 과제 3 · Array 클래스 완성 — 복사 대입, == / !=, >> 입력, explicit와 static_cast
#include <iostream>
#include <iomanip>
#include <stdexcept>
#include "task3_array_operators.h"
using namespace std;

const size_t Array::sizeDefault;

Array::Array( int arraySize )
   : size( arraySize < 0 ? 0 : arraySize ),
     ptr( size > 0 ? new int[ size ]() : nullptr )
{
}

Array::Array( const Array &arrayToCopy )
   : size( arrayToCopy.size ),
     ptr( size > 0 ? new int[ size ] : nullptr )
{
   for ( size_t i = 0; i < size; ++i )
      ptr[ i ] = arrayToCopy.ptr[ i ];
}

Array::~Array()
{
   delete[] ptr;   // new[] 와 짝
}

size_t Array::getSize() const { return size; }

// 복사 대입 연산자 — 깊은 복사
const Array &Array::operator=( const Array &right )
{
   if ( &right != this )   // 자기 대입 검사: 없으면 자기 메모리를 먼저 지워 버린다
   {
      if ( size != right.size )
      {
         delete[] ptr;                       // 기존 메모리 해제 (없으면 누수)
         size = right.size;
         ptr = size > 0 ? new int[ size ] : nullptr; // 새 크기로 할당
      }
      for ( size_t i = 0; i < size; ++i )
         ptr[ i ] = right.ptr[ i ];
   }
   return *this;   // 연쇄 대입을 위해 자기 참조 반환
}

bool Array::operator==( const Array &right ) const
{
   if ( size != right.size )
      return false;   // 원소 개수가 다르면 다르다
   for ( size_t i = 0; i < size; ++i )
      if ( ptr[ i ] != right.ptr[ i ] )
         return false;
   return true;
}

bool Array::operator!=( const Array &right ) const
{
   return !( *this == right );
}

int &Array::operator[]( int subscript )
{
   if ( subscript < 0 || subscript >= static_cast< int >( size ) )
      throw out_of_range( "Subscript out of range" );
   return ptr[ subscript ];
}

int Array::operator[]( int subscript ) const
{
   if ( subscript < 0 || subscript >= static_cast< int >( size ) )
      throw out_of_range( "Subscript out of range" );
   return ptr[ subscript ];
}

// 배열 전체를 4개씩 한 줄로 출력
ostream &operator<<( ostream &output, const Array &a )
{
   for ( size_t i = 0; i < a.size; ++i )
   {
      output << setw( 12 ) << a.ptr[ i ];
      if ( ( i + 1 ) % 4 == 0 )
         output << endl;
   }
   if ( a.size % 4 != 0 )
      output << endl;
   return output;
}

// 배열 전체를 입력받음
istream &operator>>( istream &input, Array &a )
{
   for ( size_t i = 0; i < a.size; ++i )
      input >> a.ptr[ i ];
   return input;
}

// const Array & 를 받는 함수 — explicit 때문에 int 로는 호출할 수 없다
void outputArray( const Array &arrayToOutput )
{
   cout << "함수가 받은 Array의 크기: " << arrayToOutput.getSize()
        << "\n내용:\n" << arrayToOutput;
}

int main()
{
   cout << "--- 1. 생성과 입력 ---\n";
   Array integers1( 7 );
   Array integers2;          // 기본 크기 10
   cout << "integers1의 크기: " << integers1.getSize() << endl;
   cout << "integers2의 크기: " << integers2.getSize() << endl;

   cout << "\nintegers1과 integers2에 17개의 정수를 입력하세요: ";
   cin >> integers1 >> integers2;
   cout << "\n입력 후:\nintegers1:\n" << integers1
        << "integers2:\n" << integers2;

   cout << "\n--- 2. != 연산자 ---\n";
   if ( integers1 != integers2 )
      cout << "integers1과 integers2는 다릅니다." << endl;

   cout << "\n--- 3. 복사 생성 (integers1으로 integers3 초기화) ---\n";
   Array integers3( integers1 );
   cout << "integers3의 크기: " << integers3.getSize() << "\n내용:\n" << integers3;

   cout << "\n--- 4. 대입 연산 (크기가 다른 경우) ---\n";
   cout << "integers2를 integers1에 대입합니다." << endl;
   integers1 = integers2;
   cout << "integers1:\n" << integers1 << "integers2:\n" << integers2;

   cout << "\n--- 5. 자기 대입 ---\n";
   integers1 = integers1;
   cout << "자기 대입 후 integers1:\n" << integers1;

   cout << "\n--- 6. 연쇄 대입 (integers3 = integers1 = integers2) ---\n";
   integers3 = integers1 = integers2;
   cout << "integers3:\n" << integers3;

   cout << "\n--- 7. == 연산자 ---\n";
   if ( integers1 == integers2 )
      cout << "integers1과 integers2는 같습니다." << endl;
   integers2[ 5 ] = 1000;
   if ( integers1 != integers2 )
      cout << "integers2[5]를 1000으로 바꾼 뒤 두 Array는 다릅니다." << endl;
   cout << "integers1[5] = " << integers1[ 5 ] << " (영향 없음 — 깊은 복사)" << endl;

   cout << "\n--- 8. explicit 생성자와 static_cast ---\n";
   outputArray( integers1 );
   // outputArray( 3 );               // 컴파일 오류: explicit 생성자는 암시적 변환에 쓰이지 않는다
   outputArray( static_cast< Array >( 3 ) ); // 명시적 변환은 허용된다

   cout << "\n--- 9. 범위를 벗어난 첨자 ---\n";
   try
   {
      integers1[ 100 ] = 1;
   }
   catch ( out_of_range &e )
   {
      cout << "예외 발생: " << e.what() << endl;
   }
}

4번에서 크기가 7인 integers1에 크기가 10인 integers2를 대입하니 integers1이 10개짜리로 바뀝니다. operator= 가 크기를 비교해 기존 메모리를 해제하고 새로 할당하기 때문입니다. 7번에서는 integers2[5]를 1000으로 바꿨는데 integers1[5]는 13 그대로입니다 — 대입이 깊은 복사이기 때문입니다(얕은 복사였다면 둘 다 1000이 됩니다). 8번의 explicit 이 핵심입니다. outputArray( 3 ) 은 컴파일 오류가 되지만(주석으로 남겨 두었습니다), static_cast< Array >( 3 ) 은 명시적 변환이라 허용되어 크기 3인 Array 가 만들어집니다. explicit 는 암시적 변환만 막습니다.
