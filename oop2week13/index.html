Q1. 함수 void show(istream& in)이 있습니다. 다음 중 show에 넘기면 컴파일되지 않는 객체는?

①ifstream a("in.txt");
②fstream b("in.txt");
③cin
④ofstream c("out.txt");

정답 및 해설 보기

정답 ④

ifstream은 istream을, fstream은 iostream을 거쳐 istream을 상속하므로 istream&에 묶입니다. cin도 istream 객체입니다. ofstream은 ostream만 상속하므로 istream&로 받을 수 없습니다. 파일 스트림이 표준 스트림의 파생 클래스라서, cin·cout에 쓰던 함수를 파일에도 그대로 쓸 수 있습니다. 근거: 14.2 (p4~5)

02

토론·검증 — 2인 1조로 묻고 확인하기

제공 코드 A — 순차 파일 쓰기와 읽기 (파일 src/discuss/write_read.cpp)

// write_read.cpp — 13주차 02 토론·검증 · 제공 코드 A (순차 파일 쓰기와 읽기)
// 14.3 순차 파일 만들기, 파일 열기 모드, 14.4 순차 파일 읽기
//
// 입력은 없습니다. 실행하면 같은 폴더에 clients.txt를 만들고 다시 읽습니다.
// (Visual Studio에서 디버그 실행하면 "같은 폴더"는 프로젝트 폴더입니다.)
#include <iostream>
#include <fstream>
#include <string>
using namespace std;

struct Client
{
    int account;
    const char* name;
    double balance;
};

int main()
{
    const Client data[3] = {
        { 100, "Jones", 24.98 },
        { 200, "Doe", 345.67 },
        { 300, "White", 0.0 }
    };

    // ---- 1) 쓰기 ----
    ofstream out("clients.txt", ios::out);                 // (A) 열기 모드
    if (!out) {
        cerr << "cannot open clients.txt for writing\n";
        return 1;
    }
    for (const Client& c : data)
        out << c.account << ' ' << c.name << ' ' << c.balance << '\n';
    cout << "wrote 3 records\n";
    out.close();                                           // (B) 파일 닫기

    // ---- 2) 읽기 ----
    ifstream in("clients.txt");                            // (C) 입력용으로 열기
    if (!in) {
        cerr << "cannot open clients.txt for reading\n";
        return 1;
    }
    int account;
    string name;
    double balance;
    int count = 0;
    while (in >> account >> name >> balance) {             // (D) 읽기 루프
        cout << account << ' ' << name << ' ' << balance << '\n';
        ++count;
    }
    cout << "read " << count << " records, eof=" << in.eof()
         << " fail=" << in.fail() << '\n';
    return 0;
}

// position.cpp — 13주차 02 토론·검증 · 제공 코드 B (파일 위치 포인터와 다시 읽기)
// 14.4 seekg/seekp/tellg/tellp, 슬라이드 22쪽의 clear + seekg(0), 14.5 순차 파일 갱신
//
// 입력은 없습니다. 실행하면 같은 폴더에 accounts.txt를 만들고 위치를 옮겨 가며 읽습니다.
// 레코드를 줄바꿈 없이 공백으로 이어 씁니다. 줄바꿈은 운영체제마다 저장되는 바이트 수가
// 다르지만(Windows는 2바이트), 공백은 어디서나 1바이트라 위치 값이 같게 나옵니다.
#include <iostream>
#include <fstream>
#include <string>
using namespace std;

// 다음 문자를 꺼내지 않고 보여 준다
void showNext(istream& in)
{
    int c = in.peek();
    if (c == EOF)
        cout << "EOF";
    else if (c == '\n')
        cout << "'\\n'";
    else
        cout << "'" << static_cast<char>(c) << "'";
}

// 현재 위치부터 끝까지 레코드를 읽어 이름만 출력한다
void listAll(istream& in, const char* tag)
{
    int account;
    string name;
    double balance;
    int n = 0;
    cout << "[" << tag << "]";
    while (in >> account >> name >> balance) {
        cout << ' ' << name;
        ++n;
    }
    cout << "  (" << n << " records)\n";
}

int main()
{
    // ---- 1) 파일 만들기 ----
    {
        ofstream out("accounts.txt", ios::out);
        out << "100 Jones 24.98 " << "200 Doe 345.67 " << "300 White 0.00";
        cout << "written, tellp=" << out.tellp() << '\n';
    }   // 블록이 끝나면 소멸자가 파일을 닫는다

    ifstream in("accounts.txt", ios::in);
    if (!in) {
        cerr << "cannot open accounts.txt\n";
        return 1;
    }

    // ---- 2) 위치 포인터 옮기기 ----
    in.seekg(0, ios::end);                                 // (A) 파일 끝으로
    cout << "size=" << in.tellg() << '\n';
    in.seekg(-4, ios::end);                                // (B) 끝에서 4바이트 앞
    cout << "tellg=" << in.tellg() << " next=";
    showNext(in);
    cout << '\n';
    in.seekg(4);                                           // (C) 처음에서 4바이트
    string word;
    in >> word;
    cout << "word=" << word << " tellg=" << in.tellg() << '\n';

    // ---- 3) 처음부터 두 번 읽기 (슬라이드 22쪽 방식) ----
    in.seekg(0);
    listAll(in, "pass 1");
    in.clear();                                            // (D) eofbit·failbit 지우기
    in.seekg(0);                                           // (E) 처음으로 되감기
    listAll(in, "pass 2");
    in.close();

    // ---- 4) 제자리 갱신 (14.5) ----
    fstream io("accounts.txt", ios::in | ios::out);
    io.seekp(20);                                          // (F) 둘째 레코드의 이름 자리
    io << "Kim";                                           // (G) 같은 길이(3자)로 덮어쓰기
    io.close();

    ifstream again("accounts.txt", ios::in);
    string line;
    while (getline(again, line))
        cout << "  |" << line << "|\n";
    return 0;
}

  • 취소한 뒤 booking.txt가 몇 줄이 되는지 확인하고, 파일 전체를 다시 쓰는 코드가 어디인지 짚어 설명하세요.
  • save()의 열기 모드를 ios::app으로 바꾸면 여러 번 예약·취소한 뒤 다시 실행했을 때 무엇이 잘못되는지 예측하고 실행해 확인하세요(토론 1과 비교).

개념 보충 객체의 상태는 프로그램이 끝나면 사라집니다. 그래서 시작할 때 파일에서 읽어 객체를 되살리고, 바뀔 때마다 파일에 다시 씁니다. 텍스트 순차 파일은 레코드 길이가 제각각이라 한 줄만 제자리에서 고칠 수 없으므로, 파일 전체를 새로 쓰는 것이 가장 단순하고 안전합니다. 저장 형식(시각 좌석 이름)은 쓰는 쪽(save)과 읽는 쪽(load)이 같은 약속을 지켜야 하고, 이름에 공백이 없어야 한다는 것도 그 약속의 일부입니다.
