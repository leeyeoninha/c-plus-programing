과제 1 · [난이도 하] Time 클래스 — std::format 출력과 예외를 던지는 set 함수

교재 그림 9.4~9.5의 Time 클래스를 만듭니다. 각 set 함수는 범위를 벗어난 값을 받으면 invalid_argument 예외를 던지고, 출력은 setw/setfill 대신 std::format을 사용합니다.

조건: 멤버는 unsigned int hour, minute, second · 생성자 explicit Time(int h = 0, int m = 0, int s = 0)은 세 set 함수를 호출해 검증까지 위임 · set 함수의 매개변수는 int로 받습니다(음수를 걸러내기 위함 — unsigned int로 받으면 -5가 4294967291이 되어 검사를 통과해 버립니다) · 범위를 벗어나면 throw invalid_argument("hour must be 0-23") 형식으로 던진다 · get 함수 3개와 출력 함수 2개는 모두 const 멤버 함수 · printUniversal()은 format("{:02}:{:02}:{:02}", ...)로 24시간 형식 · printStandard()는 12시간 형식과 AM/PM · main에서 시·분·초를 입력받아 try/catch로 감싸고 예외가 나면 e.what()을 출력 · 입력 실패도 검사할 것: if (!(cin >> h >> m >> s)) · 예외가 발생했을 때 객체가 어떤 상태로 남는지 관찰하고, 그 이유와 해결 방향을 주석이나 보고서에 서술할 것

// 과제 1 (하) · Time 클래스 — std::format 출력과 예외를 던지는 set 함수
#include <iostream>
#include <format>
#include <stdexcept>
using namespace std;

class Time {
public:
    explicit Time(int h = 0, int m = 0, int s = 0);
    void setHour(int h);
    void setMinute(int m);
    void setSecond(int s);
    unsigned int getHour() const;
    unsigned int getMinute() const;
    unsigned int getSecond() const;
    void printUniversal() const;   // 24시간 형식
    void printStandard() const;    // 12시간 형식
private:
    unsigned int hour;    // 0-23
    unsigned int minute;  // 0-59
    unsigned int second;  // 0-59
};

Time::Time(int h, int m, int s) {
    setHour(h);
    setMinute(m);
    setSecond(s);
}

void Time::setHour(int h) {
    if (h >= 0 && h < 24)
        hour = h;
    else
        throw invalid_argument("hour must be 0-23");
}
void Time::setMinute(int m) {
    if (m >= 0 && m < 60)
        minute = m;
    else
        throw invalid_argument("minute must be 0-59");
}
void Time::setSecond(int s) {
    if (s >= 0 && s < 60)
        second = s;
    else
        throw invalid_argument("second must be 0-59");
}

unsigned int Time::getHour() const   { return hour; }
unsigned int Time::getMinute() const { return minute; }
unsigned int Time::getSecond() const { return second; }

void Time::printUniversal() const {
    cout << format("{:02}:{:02}:{:02}", hour, minute, second);
}
void Time::printStandard() const {
    cout << format("{}:{:02}:{:02} {}",
                   ((hour == 0 || hour == 12) ? 12 : hour % 12),
                   minute, second,
                   (hour < 12 ? "AM" : "PM"));
}

int main() {
    Time t{13, 27, 6};
    cout << "24시간 형식: ";  t.printUniversal();  cout << '\n';
    cout << "12시간 형식: ";  t.printStandard();   cout << '\n';

    cout << "\n시, 분, 초를 입력하세요: ";
    int h = 0, m = 0, s = 0;
    if (!(cin >> h >> m >> s)) {          // 입력 실패 검사
        cout << "숫자를 입력해야 합니다.\n";
        return 1;
    }

    try {
        t.setHour(h);
        t.setMinute(m);
        t.setSecond(s);
        cout << "설정 성공 -> ";
        t.printUniversal();
        cout << " / ";
        t.printStandard();
        cout << '\n';
    }
    catch (const invalid_argument &e) {
        cout << "예외 발생: " << e.what() << '\n';
        cout << "현재 객체 상태 -> ";
        t.printUniversal();
        cout << '\n';
    }
    return 0;
}

실행 예시 ②③④를 나란히 보세요. ②는 첫 set에서 바로 예외가 나 객체가 그대로지만, ③은 시만, ④는 시·분까지 이미 반영된 뒤 예외가 나서 입력한 적 없는 중간 상태(10:27:06, 23:59:06)가 남습니다. 예외를 던져 잘못된 값을 막았는데도 객체는 일관되지 않은 상태가 된 것입니다. 이것을 막으려면 세 값을 먼저 모두 검사한 뒤 한꺼번에 대입하거나, setTime 하나로 묶어 검증하면 됩니다. 3주차에서 잘못된 값을 0으로 보정했던 방식과 무엇이 같고 무엇이 다른지 비교해 보세요.

과제 2 · [난이도 중] 합성(has-a) — Date를 멤버로 가지는 Employee

교재 그림 9.17~9.21의 구조를 따라 Date 클래스와, Date를 두 개 멤버로 가지는 Employee 클래스를 만듭니다. 멤버 초기화자 목록으로 멤버 객체를 초기화하고, 생성자·소멸자에 메시지를 넣어 호출 순서를 확인합니다.

조건: Date는 month, day, year를 가지며 생성자에서 월이 1~12를 벗어나면 invalid_argument 예외 · Date의 생성자·소멸자가 [Date 생성자] 7/24/1949 형태로 출력 · Employee는 firstName, lastName, birthDate, hireDate를 멤버로 가진다 (Date를 값으로 가질 것 — 포인터 금지) · Employee 생성자는 멤버 초기화자 목록만 사용하고, 초기화자를 적는 순서를 멤버 선언 순서와 일치시킨다 · print()는 const 멤버 함수이며 내부에서 Date::print()를 호출한다(따라서 Date::print()도 const여야 함) · main에서 const Employee 객체도 하나 만들어 const 객체로 print()가 호출되는지 확인 · 잘못된 월로 Date 생성을 시도해 예외를 잡아 볼 것

// 과제 2 (중) · 합성(has-a) — Date를 멤버로 가지는 Employee
#include <iostream>
#include <format>
#include <string>
#include <stdexcept>
using namespace std;

class Date {
public:
    explicit Date(int m = 1, int d = 1, int y = 1900);
    ~Date();
    void print() const;                 // 월/일/연 출력
    int getMonth() const { return month; }
private:
    int month, day, year;
};

Date::Date(int m, int d, int y) {
    if (m < 1 || m > 12)
        throw invalid_argument("month must be 1-12");
    month = m;
    day   = d;
    year  = y;
    cout << format("  [Date 생성자] {}/{}/{}\n", month, day, year);
}
Date::~Date() {
    cout << format("  [Date 소멸자] {}/{}/{}\n", month, day, year);
}
void Date::print() const {
    cout << format("{}/{}/{}", month, day, year);
}

class Employee {
public:
    Employee(const string &first, const string &last,
             const Date &dateOfBirth, const Date &dateOfHire);
    ~Employee();
    void print() const;
private:
    string firstName;
    string lastName;
    Date birthDate;      // has-a : Employee는 Date를 가진다
    Date hireDate;       // has-a
};

// 멤버 초기화자 목록으로 멤버 객체의 생성자에 값을 전달
Employee::Employee(const string &first, const string &last,
                   const Date &dateOfBirth, const Date &dateOfHire)
    : firstName(first),
      lastName(last),
      birthDate(dateOfBirth),
      hireDate(dateOfHire)
{
    cout << format("[Employee 생성자] {} {}\n", firstName, lastName);
}

Employee::~Employee() {
    cout << format("[Employee 소멸자] {} {}\n", firstName, lastName);
}

void Employee::print() const {
    cout << format("{} {}  입사일: ", lastName, firstName);
    hireDate.print();                // const 멤버 함수만 호출 가능
    cout << "  생년월일: ";
    birthDate.print();
    cout << '\n';
}

int main() {
    cout << "--- Date 객체 2개 생성 ---\n";
    Date birth{7, 24, 1949};
    Date hire{3, 12, 1988};

    cout << "\n--- Employee 생성 (복사 생성자로 멤버 객체 초기화) ---\n";
    Employee manager{"Bob", "Blue", birth, hire};

    cout << "\n--- 정보 출력 ---\n";
    manager.print();

    cout << "\n--- const 객체 ---\n";
    const Employee advisor{"Ann", "Gray", birth, hire};
    advisor.print();                 // print가 const라 호출 가능

    cout << "\n--- 잘못된 월로 Date 생성 시도 ---\n";
    try {
        Date bad{13, 1, 2000};
    }
    catch (const invalid_argument &e) {
        cout << "예외 발생: " << e.what() << '\n';
    }

    cout << "\n--- main 종료 (소멸 순서 확인) ---\n";
    return 0;
}

확인할 점 세 가지입니다. 첫째, [Date 생성자]는 두 번만 찍혔는데 [Date 소멸자]는 여섯 번 찍힙니다 — Employee가 Date를 값으로 가지므로 기본 복사 생성자로 복사본이 만들어졌고(기본 복사 생성자에는 출력문이 없어 보이지 않습니다), 그 복사본들도 각자 소멸하기 때문입니다. 둘째, Employee가 먼저 소멸하고 그 뒤에 멤버 Date가 소멸하며, 멤버는 선언의 역순으로 소멸합니다(hireDate → birthDate). 셋째, const Employee에서 print()가 호출되려면 Employee::print와 Date::print가 둘 다 const여야 합니다 — 하나만 빠뜨려도 컴파일되지 않습니다.

과제 3 · [난이도 중] 프렌드 함수·프렌드 클래스와 생성자·소멸자 호출 순서

Engine과 Car 두 클래스를 만들어, 프렌드 클래스로 클래스 전체에 권한을 주는 경우와 프렌드 함수로 함수 하나에만 권한을 주는 경우를 모두 구현합니다. 동시에 전역·자동·static·임시 객체의 생성·소멸 순서를 출력으로 확인합니다.

조건: Engine의 horsePower, running, label은 모두 private · Engine 정의 안에 friend class Car;와 friend void inspect(const Engine &e);를 선언한다(권한을 주는 쪽에 쓴다) · Car는 Engine을 멤버로 가진다(has-a) — 멤버 초기화자 목록으로 초기화 · Car::start()는 engine.running = true;처럼 private을 직접 수정 · 프렌드 함수 inspect는 멤버가 아니면서 Engine의 private을 읽어 출력 · 두 클래스 모두 생성자·소멸자에 메시지 출력 · main 밖 전역 Car, main의 자동 Car, 블록 안 Car, static Car, 그리고 임시 Engine 객체를 각각 만들어 순서를 관찰할 것

// 과제 3 (중) · 프렌드 함수/프렌드 클래스와 생성자·소멸자 호출 순서
#include <iostream>
#include <format>
#include <string>
using namespace std;

class Engine {
    friend class Car;                    // Car의 모든 멤버 함수에 접근 권한 부여
    friend void inspect(const Engine &e); // 이 함수 하나에만 접근 권한 부여
public:
    explicit Engine(int hp, const string &name);
    ~Engine();
    int getHorsePower() const { return horsePower; }
private:
    int    horsePower;
    bool   running = false;
    string label;
};

Engine::Engine(int hp, const string &name) : horsePower(hp), label(name) {
    cout << format("  [Engine 생성자] {} ({}마력)\n", label, horsePower);
}
Engine::~Engine() {
    cout << format("  [Engine 소멸자] {}\n", label);
}

// 프렌드 함수 — 멤버가 아니지만 private에 접근 가능
void inspect(const Engine &e) {
    cout << format("  점검: {} / {}마력 / 상태 {}\n",
                   e.label, e.horsePower, e.running ? "가동" : "정지");
}

class Car {
public:
    Car(const string &model, int hp);
    ~Car();
    void start();
private:
    string modelName;
    Engine engine;                       // has-a : Car는 Engine을 가진다
};

Car::Car(const string &model, int hp)
    : modelName(model), engine(hp, model + "-엔진")
{
    cout << format("[Car 생성자] {}\n", modelName);
}
Car::~Car() {
    cout << format("[Car 소멸자] {}\n", modelName);
}

void Car::start() {
    engine.running = true;               // friend class이므로 private 직접 접근
    cout << format("{} 시동 ({}마력)\n", modelName, engine.horsePower);
}

Car globalCar{"전역차", 100};             // 전역 객체

int main() {
    cout << "\nMAIN 시작\n";
    Car myCar{"세단", 250};

    cout << "\n--- 프렌드 클래스: Car가 Engine의 private 수정 ---\n";
    myCar.start();
    inspect(Engine{90, "임시엔진"});      // 임시 객체 + 프렌드 함수

    cout << "\n--- 블록 스코프 ---\n";
    {
        Car blockCar{"블록차", 130};
        blockCar.start();
    }
    cout << "블록을 벗어났습니다\n";

    static Car staticCar{"정적차", 180};  // static 지역 객체

    cout << "\nMAIN 종료\n";
    return 0;
}

읽는 법: 멤버 Engine이 Car보다 먼저 생성되고 나중에 소멸합니다(생성은 안에서 밖으로, 소멸은 밖에서 안으로). 임시 Engine 객체는 그 문장이 끝나는 즉시 소멸합니다. 정적차는 블록을 벗어난 뒤에 생성되었는데도 자동 객체인 세단보다 나중에 소멸합니다 — static은 main 종료 시점에 소멸하기 때문입니다. 그리고 inspect에 넘긴 임시 엔진의 상태가 「정지」인 것에 주목하세요. 바로 앞에서 myCar.start()로 시동을 걸었지만 그것은 세단의 엔진이고, 임시엔진은 별개의 객체입니다.
